## WHAT YOU ARE BUILDING

A dark-humor web app where users input a life decision and get an AI-generated outcome timeline, a mood graph, and a Regret Score. Think: decision simulator meets roast machine meets social leaderboard.

**One-liner:** "You type a bad decision. We tell you exactly how badly it ruins your life."

## SUPABASE SCHEMA

Run this in the Supabase SQL editor to create all tables:

```sql
-- Users (extends Supabase auth.users)
create table public.profiles (
  id uuid references auth.users(id) primary key,
  username text unique not null,
  total_regret_score integer default 0,
  decision_count integer default 0,
  created_at timestamp with time zone default now()
);

-- Habit counts per user per category
create table public.habit_counts (
  id uuid default gen_random_uuid() primary key,
  user_id uuid references public.profiles(id) on delete cascade,
  category text not null,
  count integer default 1,
  unique(user_id, category)
);

-- All decisions simulated
create table public.decisions (
  id uuid default gen_random_uuid() primary key,
  user_id uuid references public.profiles(id) on delete set null,
  decision_text text not null,
  personality_mode text check (personality_mode in ('savage', 'motivational', 'reality')),
  regret_score integer check (regret_score between 0 and 100),
  timeline jsonb not null,        -- { day1, week1, month1, year1 }
  mood_data integer[] not null,   -- 5 values 0-100
  ai_roast text not null,
  category text not null,
  created_at timestamp with time zone default now()
);

-- Leaderboard entries
create table public.leaderboard_entries (
  id uuid default gen_random_uuid() primary key,
  decision_id uuid references public.decisions(id) on delete cascade,
  user_id uuid references public.profiles(id) on delete cascade,
  username text not null,
  decision_text text not null,
  regret_score integer not null,
  ai_roast text not null,
  skull_count integer default 0,
  fire_count integer default 0,
  cry_count integer default 0,
  relate_count integer default 0,
  created_at timestamp with time zone default now()
);

-- Response cache (avoid duplicate Groq calls, save rate limit)
create table public.response_cache (
  id uuid default gen_random_uuid() primary key,
  cache_key text unique not null,   -- normalized decision text + mode
  response jsonb not null,
  created_at timestamp with time zone default now()
);

-- Enable Row Level Security
alter table public.profiles enable row level security;
alter table public.decisions enable row level security;
alter table public.leaderboard_entries enable row level security;
alter table public.habit_counts enable row level security;

-- Policies
create policy "Public leaderboard read" on public.leaderboard_entries for select using (true);
create policy "Users read own decisions" on public.decisions for select using (auth.uid() = user_id);
create policy "Users insert own decisions" on public.decisions for insert with check (auth.uid() = user_id);
create policy "Users read own profile" on public.profiles for select using (auth.uid() = id);
create policy "Users update own profile" on public.profiles for update using (auth.uid() = id);
```

## API ENDPOINTS — BUILD ALL OF THESE

### POST `/api/simulate`
**Auth:** optional (works for guests too)
**Body:**
```json
{ "decision": "Skip gym today", "mode": "savage" }
```
**Response:**
```json
{
  "timeline": { "day1": "...", "week1": "...", "month1": "...", "year1": "..." },
  "moodData": [85, 70, 50, 30, 15],
  "regretScore": 62,
  "aiRoast": "...",
  "category": "gym"
}
```
**Logic:**
1. Check `response_cache` for `(normalizedDecision + mode)` — if hit and < 24h old, return it.
2. If user authenticated, fetch `habit_counts` to get `repeatCount`, else `repeatCount = 0`.
3. Call Groq with prompt template below.
4. Save to `decisions` table and upsert `response_cache`.
5. If authenticated, upsert `habit_counts` and increment `profiles.total_regret_score`.

---

### GET `/api/leaderboard`
**Query params:** `?tab=alltime|weekly|relatable&limit=20&page=1`
Sort: `alltime` → regret_score DESC | `weekly` → last 7 days, regret_score DESC | `relatable` → relate_count DESC

### POST `/api/leaderboard/submit`
**Auth:** required
**Body:** `{ decisionId: string }`
Limit: max 3 submissions per user per day. Basic keyword filter for explicit content before inserting.

### POST `/api/leaderboard/:id/react`
**Auth:** required
**Body:** `{ reaction: "skull" | "fire" | "cry" | "relate" }`
Increments the relevant count column.

### GET `/api/user/stats`
**Auth:** required
Returns profile + habit counts + badge eligibility.

### GET `/api/health`
Returns `{ status: "ok" }`

## GROQ INTEGRATION — `server/services/groq.ts`

```bash
npm install groq-sdk
```

```ts
import Groq from "groq-sdk"

const groq = new Groq({ apiKey: process.env.GROQ_API_KEY })

const systemPrompts = {
  savage: `You are a brutally honest, sarcastic life coach who roasts people's bad decisions with dark humor.
You speak like a disappointed older sibling who has seen everything. Never sugarcoat.
Tone: savage, funny, PG-13 max. No slurs or genuinely harmful content.
IMPORTANT: Respond ONLY with valid JSON. No markdown, no explanation, no extra text whatsoever.`,

  motivational: `You are an aggressively positive life coach whose optimism becomes unhinged as consequences mount.
Day 1 you're cheering. By Year 1 you're clearly in denial. The humor is in the contrast.
Tone: fake-positive, ironic, darkly funny.
IMPORTANT: Respond ONLY with valid JSON. No markdown, no explanation, no extra text whatsoever.`,

  reality: `You are a calm, clinical analyst. No jokes. Cold, factual consequences only.
Use data, percentages, and medical/psychological terminology where relevant.
Tone: dry, factual, slightly ominous.
IMPORTANT: Respond ONLY with valid JSON. No markdown, no explanation, no extra text whatsoever.`
}

const buildUserPrompt = (decision: string, repeatCount: number) => `
Decision: "${decision}"
Times this category of decision has been made before: ${repeatCount}

Respond ONLY with this exact JSON:
{
  "timeline": {
    "day1": "1-2 sentences",
    "week1": "1-2 sentences",
    "month1": "1-2 sentences",
    "year1": "1-2 sentences"
  },
  "moodData": [number, number, number, number, number],
  "regretScore": number,
  "aiRoast": "one punchy line",
  "category": "gym|food|sleep|relationship|study|work|finance|social|other"
}

Rules:
- moodData: 5 integers 0-100. Generally declining. Savage = steep cliff. Motivational = small rise then crash. Reality = steady slope down.
- regretScore: integer 0-100. Use the full range creatively — not everything is 55-70.
- If repeatCount >= 3, call it out directly in aiRoast. Escalate the intensity with each threshold.
- aiRoast = the single most devastating or funniest line. Make it memorable.
`

export async function generateRegretResponse(
  decision: string,
  mode: "savage" | "motivational" | "reality",
  repeatCount: number
) {
  try {
    const completion = await groq.chat.completions.create({
      model: "llama-3.3-70b-versatile",
      messages: [
        { role: "system", content: systemPrompts[mode] },
        { role: "user", content: buildUserPrompt(decision, repeatCount) }
      ],
      temperature: 0.85,
      max_tokens: 600,
    })

    const raw = completion.choices[0].message.content ?? ""
    const cleaned = raw.replace(/```json|```/g, "").trim()
    const parsed = JSON.parse(cleaned)

    // Validate required fields
    if (!parsed.timeline || !parsed.moodData || parsed.regretScore === undefined) {
      throw new Error("Invalid response shape from Groq")
    }

    return parsed
  } catch (err) {
    console.error("Groq error, using fallback:", err)
    return getFallbackResponse(mode)   // implement using fallbacks.json
  }
}
```

```ts
// After Groq responds in simulate route:
const escalationSuffix: Record<number, string> = {
  3:  "Third time. We're noticing a pattern.",
  5:  "Fifth time. This isn't a mistake anymore — it's a lifestyle.",
  7:  "Seventh time. You are being scientifically studied.",

const threshold = [10, 7, 5, 3].find(t => repeatCount >= t)
if (threshold) {
  response.aiRoast += ` ${escalationSuffix[threshold]}`
}

// Upsert habit_counts after:

```

## THINGS TO HANDLE IN CODE

1. **Groq JSON parsing** — strip ` ```json ``` ` fences before `JSON.parse()`. Always wrap in try/catch. Fall back to `fallbacks.json` on any error.
2. **Groq rate limits** — cache aggressively. Same normalized decision + mode = serve cache, never hit Groq again for 24h.
3. **Supabase RLS** — test that users can only access their own data. The schema includes policies but verify after setup.
4. **Leaderboard spam** — before inserting, count user's `leaderboard_entries` in last 24h. If >= 3, reject with 429.
