## 2. CODING STANDARDS & RULES

### 2.1 TypeScript Rules

```
- Strict mode ON — tsconfig: "strict": true
- No `any` — ever. Use `unknown` and narrow it.
- No non-null assertions (!) unless absolutely unavoidable with a comment explaining why
- All function parameters and return types must be explicitly typed
- Use `interface` for object shapes, `type` for unions/aliases
- Enums: use `as const` objects instead of TypeScript enums
```

```ts
// WRONG
const mode: any = getMode()
const user = data!.user

// RIGHT
const mode: SimulationMode = getMode()
const user = data?.user ?? null
```

### 2.2 React Rules

```
- Functional components only — no class components
- Named exports only — no default exports on components
- useEffect dependencies must be complete — no eslint-disable comments to hide warnings
- Never mutate state directly — always use setter functions
- No inline functions in JSX that do heavy work — extract to named handlers
- Keys in lists: always use stable unique IDs, never array index
- Avoid prop drilling beyond 2 levels — use Zustand store instead
```

```tsx
// WRONG
{decisions.map((d, index) => <Card key={index} ... />)}

// RIGHT
{decisions.map((d) => <Card key={d.id} ... />)}
```

### 2.3 Async / Data Fetching Rules

```
- All async operations: always use try/catch — never let promises go unhandled
- Loading states: always track with boolean, show skeleton UI
- Error states: always handle, show user-friendly message (not raw error)
- Never fetch in useEffect without cleanup or AbortController
- All API calls go through lib/api.ts — no raw fetch() calls in components
```

```ts
// WRONG — fetch inside component
useEffect(() => {
  fetch('/api/simulate').then(r => r.json()).then(setData)
}, [])

// RIGHT — use the api lib + proper state
const { data, loading, error } = useSimulate()
```

### 2.4 Import Rules

```
- Use path aliases — never relative `../../` hell
- Alias: @/ maps to client/src/
- External imports before internal imports, blank line between
- Destructure named imports — no namespace imports (import * as X)
```

```ts
// WRONG
import { something } from '../../../lib/utils'
import * as Recharts from 'recharts'

// RIGHT
import { LineChart, Line } from 'recharts'
import { getScoreColor } from '@/lib/utils'
```

Configure in `vite.config.ts` and `tsconfig.json`:
```ts
// vite.config.ts
resolve: { alias: { '@': path.resolve(__dirname, './src') } }

// tsconfig.json
"paths": { "@/*": ["./src/*"] }
```

### 2.5 Styling Rules

```
- Tailwind utility classes only — no inline style={{ }} unless for dynamic values (e.g. SVG coordinates)
- No separate CSS files per component — index.css is for base resets only
- Use custom Tailwind tokens from tailwind.config.js (e.g. bg-bg-primary, not bg-[#0F0E17])
- No !important
- No styled-components, no CSS modules, no emotion
- Dynamic classes: use clsx or cn() helper — never string template literals
```

```tsx
// WRONG
<div style={{ background: '#0F0E17' }}>
<div className={`card ${isActive ? 'active' : ''}`}>

// RIGHT
<div className="bg-bg-primary">
<div className={cn('card', isActive && 'ring-2 ring-accent-purple')}>
```

Install `clsx` + `tailwind-merge`, create a `cn()` helper in `lib/utils.ts`:
```ts
import { clsx, type ClassValue } from 'clsx'
import { twMerge } from 'tailwind-merge'

export function cn(...inputs: ClassValue[]) {
  return twMerge(clsx(inputs))
}
```

### 2.6 Naming Conventions

| Thing | Convention |
|---|---|
| Variables | `camelCase` |
| Functions | `camelCase` |
| Components | `PascalCase` |
| Types/Interfaces | `PascalCase` |
| Constants | `SCREAMING_SNAKE_CASE` |
| Zustand store fields | `camelCase` |
| Database columns | `snake_case` (Supabase/PostgreSQL standard) |
| API endpoints | `kebab-case` e.g. `/api/user-stats` |
| Env variables | `SCREAMING_SNAKE_CASE` |

### 2.7 Comments & Documentation

```
- Only comment WHY, not WHAT — the code shows what, comments explain why
- All exported functions must have a JSDoc comment
- No commented-out code in commits — delete it
- TODO comments: format as // TODO(yourname): description
```

```ts
// WRONG
// loop through decisions
decisions.forEach(d => process(d))

// RIGHT — no comment needed, code is self-explanatory
decisions.forEach(processDecision)

// RIGHT — comment explains WHY
// Normalize before cache lookup to catch "Skip Gym" and "skip gym" as the same key
const cacheKey = normalizeDecision(decision) + ':' + mode
```

## 6. ERROR HANDLING PATTERNS

### 6.1 The Golden Rule

**Every async operation must have a try/catch. Every error must result in a user-visible state or a logged server error. No silent failures.**

### 6.2 Client-Side Error Pattern

```ts
// Standard pattern for all API calls
const [data, setData] = useState<SimulationResult | null>(null)
const [loading, setLoading] = useState(false)
const [error, setError] = useState<string | null>(null)

async function handleSubmit(decision: string, mode: SimulationMode) {
  setLoading(true)
  setError(null)

  try {
    const result = await api.simulate(decision, mode)
    setData(result)
  } catch (err) {
    // Never show raw error messages to users
    setError('Something went wrong. Please try again.')
    // Log the real error for debugging
    console.error('[handleSubmit]', err)
  } finally {
    setLoading(false)
  }
}
```

### 6.3 Server-Side Error Pattern

All Express routes use this structure:

```ts
// routes/simulate.ts
router.post('/simulate', async (req, res) => {
  try {
    const { decision, mode } = req.body

    // Validate input
    if (!decision || typeof decision !== 'string') {
      return res.status(400).json({ error: 'decision is required and must be a string' })
    }
    if (!['savage', 'motivational', 'reality'].includes(mode)) {
      return res.status(400).json({ error: 'mode must be savage, motivational, or reality' })
    }

    const result = await generateRegretResponse(decision, mode, 0)
    return res.status(200).json(result)

  } catch (err) {
    console.error('[POST /api/simulate]', err)
    return res.status(500).json({ error: 'Internal server error' })
  }
})
```

### 6.4 Groq Error Handling

```ts
// services/groq.ts
export async function generateRegretResponse(
  decision: string,
  mode: SimulationMode,
  repeatCount: number
): Promise<SimulationResult> {
  try {
    const completion = await groq.chat.completions.create({ ... })
    const raw = completion.choices[0].message.content ?? ''

    // Strip markdown fences — Llama sometimes adds them despite instructions
    const cleaned = raw.replace(/```json|```/g, '').trim()

    let parsed: unknown
    try {
      parsed = JSON.parse(cleaned)
    } catch {
      console.error('[groq] JSON parse failed, raw response:', raw)
      throw new Error('Invalid JSON from Groq')
    }

    // Validate shape before trusting it
    if (!isValidSimulationResult(parsed)) {
      console.error('[groq] Invalid shape:', parsed)
      throw new Error('Unexpected response shape from Groq')
    }

    return parsed

  } catch (err) {
    // Log the error but don't crash — use fallback
    console.error('[groq] Falling back to pre-written response:', err)
    return getFallbackResponse(mode)
  }
}

// Type guard for validation
function isValidSimulationResult(data: unknown): data is SimulationResult {
  if (typeof data !== 'object' || data === null) return false
  const d = data as Record<string, unknown>
  return (
    typeof d.timeline === 'object' &&
    Array.isArray(d.moodData) &&
    d.moodData.length === 5 &&
    typeof d.regretScore === 'number' &&
    d.regretScore >= 0 &&
    d.regretScore <= 100 &&
    typeof d.aiRoast === 'string' &&
    typeof d.category === 'string'
  )
}
```

## 7. API & DATABASE CONVENTIONS

### 7.1 API Design Rules

```
- All routes prefixed with /api/
- Use nouns not verbs in route paths
- Use HTTP methods correctly (GET=read, POST=create, PATCH=update, DELETE=delete)
- Request bodies: always JSON (Content-Type: application/json)
- Always validate request body before using it
- Always return consistent response shapes (see below)
- No sensitive data in query strings (tokens, passwords, keys)
```

### 7.2 Response Shapes

**Success:**
```ts
// Single resource
{ "data": { ... } }

// List of resources
{ "data": [...], "count": number, "page": number }

// Action with no return data
{ "success": true }
```

**Error:**
```ts
{ "error": "Human-readable description" }
```

### 7.3 Request Validation

Validate all incoming request bodies before using them:

```ts
// Simple manual validation (no extra library needed for MVP)
function validateSimulateBody(body: unknown): { decision: string; mode: SimulationMode } {
  if (typeof body !== 'object' || body === null) {
    throw new ValidationError('Request body must be a JSON object')
  }

  const { decision, mode } = body as Record<string, unknown>

  if (!decision || typeof decision !== 'string' || decision.trim().length === 0) {
    throw new ValidationError('decision must be a non-empty string')
  }

  if (!['savage', 'motivational', 'reality'].includes(mode as string)) {
    throw new ValidationError('mode must be savage, motivational, or reality')
  }

  if (decision.trim().length > 500) {
    throw new ValidationError('decision must be under 500 characters')
  }

  return { decision: decision.trim(), mode: mode as SimulationMode }
}
```

### 7.4 Authentication Middleware

```ts
// middleware/auth.ts
import { createClient } from '@supabase/supabase-js'

export async function requireAuth(req: Request, res: Response, next: NextFunction) {
  const token = req.headers.authorization?.replace('Bearer ', '')

  if (!token) {
    return res.status(401).json({ error: 'Authorization token required' })
  }

  try {
    const supabase = createClient(
      process.env.SUPABASE_URL!,
      process.env.SUPABASE_SERVICE_KEY!
    )

    const { data: { user }, error } = await supabase.auth.getUser(token)

    if (error || !user) {
      return res.status(401).json({ error: 'Invalid or expired token' })
    }

    req.user = user  // attach to request for downstream use
    next()
  } catch {
    return res.status(401).json({ error: 'Token verification failed' })
  }
}

// Usage on protected routes
router.post('/leaderboard/submit', requireAuth, async (req, res) => { ... })
```

### 7.5 Rate Limiting

```ts

import rateLimit from 'express-rate-limit'

// General API rate limit
export const apiLimiter = rateLimit({
  windowMs: 60 * 1000,          // 1 minute window
  max: 30,                      // 30 requests per minute per IP
  message: { error: 'Too many requests. Slow down.' },
  standardHeaders: true,
  legacyHeaders: false,

// Stricter limit for simulate (AI calls are expensive)
export const simulateLimiter = rateLimit({
  windowMs: 60 * 1000,
  max: 10,                      // 10 simulations per minute per IP
  message: { error: 'Simulation rate limit reached. Wait a moment.' },
})

// Apply in index.ts
app.use('/api', apiLimiter)
app.use('/api/simulate', simulateLimiter)
```

```ts
}))
```

### 7.7 Supabase Database Conventions

```
- All table names: plural snake_case (decisions, leaderboard_entries, habit_counts)
- All column names: snake_case
- Primary keys: always uuid using gen_random_uuid()
- Foreign keys: always name as {referenced_table_singular}_id (e.g. user_id, decision_id)
- Timestamps: created_at and updated_at on all tables, type timestamptz
- Soft deletes: not used in MVP — hard delete is fine
- Row Level Security: enabled on all tables — never disabled
- All queries go through Supabase client — no raw SQL in routes
```

### 7.8 Supabase Query Conventions

```ts
// Always use the typed Supabase client with generated types (optional but preferred)
// Always select only the columns you need — no SELECT *

// WRONG
const { data } = await supabase.from('decisions').select('*')

// RIGHT
const { data } = await supabase
  .from('decisions')
  .select('id, decision_text, regret_score, created_at')
  .eq('user_id', userId)
  .order('created_at', { ascending: false })
  .limit(20)

// Upsert pattern for habit_counts
await supabase
  .from('habit_counts')
  .upsert(
    { user_id: userId, category, count: 1 },
    { onConflict: 'user_id,category', ignoreDuplicates: false }
  )
// Note: this won't increment — for increment use RPC or read-modify-write
```

### 7.9 Response Cache Convention

Cache key format:
```ts
// Normalize decision text before creating cache key
function normalizeDecision(text: string): string {
  return text
    .toLowerCase()
    .trim()
    .replace(/[^a-z0-9\s]/g, '')   // remove punctuation
    .replace(/\s+/g, ' ')           // collapse whitespace
}

const cacheKey = `${normalizeDecision(decision)}:${mode}`
// e.g. "skip gym today:savage"
// "Skip Gym Today!" and "skip gym today" resolve to same cache key
```

Cache lookup pattern:
```ts
// Check cache first
const { data: cached } = await supabase
  .from('response_cache')
  .select('response, created_at')
  .eq('cache_key', cacheKey)
  .single()

if (cached) {
  const ageMs = Date.now() - new Date(cached.created_at).getTime()
  const MAX_AGE_MS = 24 * 60 * 60 * 1000  // 24 hours
  if (ageMs < MAX_AGE_MS) {
    return cached.response as SimulationResult
  }
}

// Cache miss — call Groq
const result = await generateRegretResponse(decision, mode, repeatCount)

// Save to cache (upsert — update if key exists)
await supabase
  .from('response_cache')
  .upsert({ cache_key: cacheKey, response: result })

return result
```
